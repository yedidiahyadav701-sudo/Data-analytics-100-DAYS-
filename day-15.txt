Questions of Queries
=================
(A) Single Row Subqueries
—----------------------------------------------------------
-- 1. Employee with salary greater than average salary
SELECT emp_name, salary
FROM Employee
WHERE salary > (SELECT AVG(salary) FROM Employee);

-- 2. Employee(s) in the same department as Ravi
SELECT emp_name
FROM Employee
WHERE dept_id = (SELECT dept_id FROM Employee WHERE emp_name = 'Ravi');

-- 3. Employee(s) with salary equal to minimum salary
SELECT emp_name, salary
FROM Employee
WHERE salary = (SELECT MIN(salary) FROM Employee);

-- 4. Department with highest budget
SELECT dept_name
FROM Department
WHERE dept_id = (SELECT dept_id FROM Project ORDER BY budget DESC LIMIT 1);

-- 5. Employee with earliest hire date
SELECT emp_name, hire_date
FROM Employee
WHERE hire_date = (SELECT MIN(hire_date) FROM Employee);

-- 6. Employee(s) with salary equal to maximum salary
SELECT emp_name, salary
FROM Employee
WHERE salary = (SELECT MAX(salary) FROM Employee);

-- 7. Department of employee ‘Priya’
SELECT dept_name
FROM Department
WHERE dept_id = (SELECT dept_id FROM Employee WHERE emp_name = 'Priya');

 Multi-Row Subqueries with IN
1. Employees working in departments that have projects
SELECT emp_name, dept_id
FROM Employee
WHERE dept_id IN (SELECT dept_id FROM Project);


2. Employees working in departments located in Bangalore
SELECT emp_name, dept_id
FROM Employee
WHERE dept_id IN (SELECT dept_id FROM Department WHERE location = 'Bangalore');


3. Employees who earn the same salary as someone in HR
SELECT emp_name, salary
FROM Employee
WHERE salary IN (SELECT salary FROM Employee WHERE dept_id = 1);


4. Projects belonging to departments where Ravi works
SELECT proj_name, budget
FROM Project
WHERE dept_id IN (SELECT dept_id FROM Employee WHERE emp_name = 'Ravi');


5. Employees in departments that have budget more than 10 lakhs
SELECT emp_name, dept_id
FROM Employee
WHERE dept_id IN (SELECT dept_id FROM Project WHERE budget > 1000000);


6. Employees who belong to the same department as Sneha
SELECT emp_name, dept_id
FROM Employee
WHERE dept_id IN (SELECT dept_id FROM Employee WHERE emp_name = 'Sneha');


7. Departments that have employees earning more than 80k
SELECT dept_name
FROM Department
WHERE dept_id IN (SELECT dept_id FROM Employee WHERE salary > 80000);



(B) Multi-Row Subqueries – ANY / SOME
—-----------------------------------------------------------------------------------
-- 1. Employees with salary greater than ANY IT employee
SELECT emp_name, salary
FROM Employee
WHERE salary > ANY (SELECT salary FROM Employee WHERE dept_id = 2);

-- 2. Employees hired before ANY Finance employee
SELECT emp_name, hire_date
FROM Employee
WHERE hire_date < ANY (SELECT hire_date FROM Employee WHERE dept_id = 3);

-- 3. Employees in same department as ANY employee earning > 80000
SELECT emp_name, dept_id
FROM Employee
WHERE dept_id = ANY (SELECT dept_id FROM Employee WHERE salary > 80000);

-- 4. Employees earning more than ANY Marketing employee
SELECT emp_name, salary
FROM Employee
WHERE salary > ANY (SELECT salary FROM Employee WHERE dept_id = 4);

-- 5. Projects with budget greater than ANY IT project
SELECT proj_name, budget
FROM Project
WHERE budget > ANY (SELECT budget FROM Project WHERE dept_id = 2);

-- 6. Employees whose salary < ANY HR employee salary
SELECT emp_name, salary
FROM Employee
WHERE salary < ANY (SELECT salary FROM Employee WHERE dept_id = 1);

-- 7. Employees who joined before ANY HR employee
SELECT emp_name, hire_date
FROM Employee
WHERE hire_date < ANY (SELECT hire_date FROM Employee WHERE dept_id = 1);



(C) Multi-Row Subqueries – ALL
—-------------------------------------------------------------------

SELECT emp_name, salary

SELECT emp_name, salary

SELECT proj_name, budget

-- 4. Employees hired before ALL Marketing employees
SELECT emp_name, hire_date
FROM Employee

(D) Nested Subqueries
—----------------------------------------------
-- 1. Employees in dept having highest budget project
SELECT emp_name
FROM Employee
WHERE dept_id = (SELECT dept_id FROM Project WHERE budget = 
                 (SELECT MAX(budget) FROM Project));

-- 2. Employees earning more than average salary of IT dept
SELECT emp_name, salary
FROM Employee
WHERE salary > (SELECT AVG(salary) FROM Employee WHERE dept_id = 
                (SELECT dept_id FROM Department WHERE dept_name = 'IT'));

-- 3. Employees in dept of employee with highest salary
SELECT emp_name
FROM Employee
WHERE dept_id = (SELECT dept_id FROM Employee WHERE salary = 
                (SELECT MAX(salary) FROM Employee));

-- 4. Projects belonging to dept of employee ‘Sneha’
SELECT proj_name
FROM Project
WHERE dept_id = (SELECT dept_id FROM Employee WHERE emp_name = 'Sneha');

-- 5. Employees whose dept has budget < average project budget
SELECT emp_name
FROM Employee
WHERE dept_id IN (SELECT dept_id FROM Project 
                  WHERE budget < (SELECT AVG(budget) FROM Project));

-- 6. Employees in location ‘Delhi’
SELECT emp_name
FROM Employee

(E) Correlated Subqueries
—-------------------------------------------------------
-- 1. Employees earning more than avg salary of their dept
SELECT emp_name, salary
FROM Employee e
WHERE salary > (SELECT AVG(salary) FROM Employee WHERE dept_id = e.dept_id);

-- 2. Employees hired before min hire_date of their dept
SELECT emp_name, hire_date
FROM Employee e
WHERE hire_date = (SELECT MIN(hire_date) FROM Employee WHERE dept_id = e.dept_id);

-- 3. Departments with employee count > 2
SELECT dept_name
FROM Department d
WHERE 2 < (SELECT COUNT(*) FROM Employee e WHERE e.dept_id = d.dept_id);

-- 4. Projects with budget greater than avg budget of their dept
SELECT proj_name, budget
FROM Project p
WHERE budget > (SELECT AVG(budget) FROM Project WHERE dept_id = p.dept_id);

-- 5. Employees earning highest salary in their dept
SELECT emp_name, salary, dept_id
FROM Employee e
WHERE salary = (SELECT MAX(salary) FROM Employee WHERE dept_id = e.dept_id);

-- 6. Employees earning less than avg salary of their dept
SELECT emp_name, salary
FROM Employee e
WHERE salary < (SELECT AVG(salary) FROM Employee WHERE dept_id = e.dept_id);

Emptyset
—-------
-- 7. Departments having budget > all project budgets of IT dept
SELECT dept_name
FROM Department d
WHERE EXISTS (SELECT 1 FROM Project p 
              WHERE p.dept_id = d.dept_id 
              AND p.budget > ALL (SELECT budget FROM Project WHERE dept_id = 2));


(F) EXISTS / NOT EXISTS
—---------------------------------------------------
-- 1. Departments having employees
SELECT dept_name
FROM Department d
WHERE EXISTS (SELECT 1 FROM Employee e WHERE e.dept_id = d.dept_id);

Emptyset
—-------------
-- 2. Departments without employees
SELECT dept_name
FROM Department d
WHERE NOT EXISTS (SELECT 1 FROM Employee e WHERE e.dept_id = d.dept_id);

-- 3. Employees who are assigned to projects
SELECT emp_name
FROM Employee e
WHERE EXISTS (SELECT 1 FROM Project p WHERE p.dept_id = e.dept_id);
Emptyset
—-------------
-- 4. Employees not in any project dept
SELECT emp_name
FROM Employee e
WHERE NOT EXISTS (SELECT 1 FROM Project p WHERE p.dept_id = e.dept_id);

-- 5. Projects in departments having more than 1 employee
SELECT proj_name
FROM Project p
WHERE EXISTS (SELECT 1 FROM Employee e WHERE e.dept_id = p.dept_id HAVING COUNT(*) > 1);

-- 6. Departments having projects with budget > 5 lakhs
SELECT dept_name
FROM Department d
WHERE EXISTS (SELECT 1 FROM Project p WHERE p.dept_id = d.dept_id AND p.budget > 500000);
Emptyset
—-------------
-- 7. Employees working in depts without any project
SELECT emp_name
FROM Employee e
WHERE NOT EXISTS (SELECT 1 FROM Project p WHERE p.dept_id = e.dept_id);

-- 1. CREATE TABLES

CREATE TABLE country (
    Code            VARCHAR(3) PRIMARY KEY,
    Name            VARCHAR(100) NOT NULL,
    Continent       VARCHAR(30)  NOT NULL,
    Region          VARCHAR(50)  NOT NULL,
    Population      BIGINT       NOT NULL,
    LifeExpectancy  DECIMAL(4,1),
    GNP             DECIMAL(15,2),
    Capital         INT
);

CREATE TABLE city (
    ID           INT PRIMARY KEY AUTO_INCREMENT,
    Name         VARCHAR(100) NOT NULL,
    CountryCode  VARCHAR(3)   NOT NULL,
    District     VARCHAR(50),
    Population   BIGINT       NOT NULL
);

CREATE TABLE countrylanguage (
    CountryCode  VARCHAR(3)  NOT NULL,
    Language     VARCHAR(50) NOT NULL,
    IsOfficial   ENUM('T','F') NOT NULL,
    Percentage   DECIMAL(4,1)
);

----------------------------------------------------------------------
DAY 21: SUBQUERIES — 1. SINGLE ROW SUBQUERIES
----------------------------------------------------------------------

Examples:
Example 1: Find the country with the largest population
SELECT Name, Population
FROM country
WHERE Population = (SELECT MAX(Population) FROM country);

Example 2: Find cities in the same country as 'New York'
SELECT Name, Population
FROM city
WHERE CountryCode = (SELECT CountryCode FROM city WHERE Name = 'New York');

Example 3: Find countries with population greater than India's population
SELECT Name, Population
FROM country
WHERE Population > (SELECT Population FROM country WHERE Name = 'India');

Example 4: Find the capital city of Brazil
SELECT c.Name as CityName, c.Population
FROM city c
WHERE c.ID = (SELECT Capital FROM country WHERE Name = 'Brazil');

Example 5: Find countries in the same continent as Japan
SELECT Name, Continent
FROM country
WHERE Continent = (SELECT Continent FROM country WHERE Name = 'Japan');


----------------------------------------------------------------------
DAY 21: SUBQUERIES — 2. MULTI ROW SUBQUERIES WITH ANY/SOME
----------------------------------------------------------------------
Examples:
Example 1: Find countries with population greater than ANY European country
SELECT Name, Population, Continent
FROM country
WHERE Population > ANY (SELECT Population FROM country WHERE Continent = 'Europe')
  AND Continent != 'Europe';

Example 2: Find cities with population greater than ANY city in USA
SELECT Name, Population, CountryCode
FROM city                  	
WHERE Population > ANY (SELECT Population FROM city WHERE CountryCode = 'USA')
  AND CountryCode != 'USA';

Example 3: Using =ANY (equivalent to IN)
SELECT Name, Population
FROM city
WHERE CountryCode = ANY (SELECT Code FROM country WHERE Continent = 'Asia');
 
-- This is equivalent to:
SELECT Name, Population
FROM city
WHERE CountryCode IN (SELECT Code FROM country WHERE Continent = 'Asia');

Example 4: Find countries with life expectancy greater than ANY African country
SELECT Name, LifeExpectancy, Continent
FROM country
WHERE LifeExpectancy > ANY (SELECT LifeExpectancy
                       	FROM country
                       	WHERE Continent = 'Africa' AND LifeExpectancy IS NOT NULL)
  AND Continent != 'Africa';

Example 5: Cities larger than ANY capital city
SELECT c.Name, c.Population, co.Name as CountryName
FROM city c
JOIN country co ON c.CountryCode = co.Code
WHERE c.Population > ANY (SELECT ci.Population
                     	FROM city ci
                     	JOIN country co2 ON ci.ID = co2.Capital
                     	WHERE ci.Population IS NOT NULL);


----------------------------------------------------------------------
: SUBQUERIES — 3. MULTI ROW SUBQUERIES WITH ALL
----------------------------------------------------------------------

Examples:
Example 1: Find countries with population greater than ALL European countries
SELECT Name, Population, Continent
FROM country
WHERE Population > ALL (SELECT Population FROM country WHERE Continent = 'Europe');

Example 2: Find the smallest city (population-wise) that is still larger than ALL cities in Monaco
SELECT Name, Population, CountryCode
FROM city
WHERE Population > ALL (SELECT Population FROM city WHERE CountryCode = 'MCO')
ORDER BY Population
LIMIT 1;

Example 3: Find countries with life expectancy greater than ALL African countries
SELECT Name, LifeExpectancy, Continent
FROM country
WHERE LifeExpectancy > ALL (SELECT LifeExpectancy
                       	FROM country
                       	WHERE Continent = 'Africa' AND LifeExpectancy IS NOT NULL);

Example 4: Find languages spoken in more countries than ALL languages in South America
SELECT Language, COUNT(*) as CountryCount
FROM countrylanguage
GROUP BY Language
HAVING COUNT(*) > ALL (SELECT COUNT(*)
                   	FROM countrylanguage cl
                   	JOIN country c ON cl.CountryCode = c.Code
                   	WHERE c.Continent = 'South America'
                   	GROUP BY cl.Language);

Example 5: Cities with population smaller than ALL capital cities
SELECT c.Name, c.Population
FROM city c
WHERE c.Population < ALL (SELECT ci.Population
                     	FROM city ci
 	                    JOIN country co ON ci.ID = co.Capital
                     	WHERE ci.Population IS NOT NULL)
ORDER BY c.Population DESC;

SELECT c.Name as CityName, c.Population, co.Name as CountryName

SELECT cl.Language, SUM(c.Population * cl.Percentage / 100) as TotalSpeakers

                    WHERE GNP = (SELECT MAX(GNP) FROM country WHERE GNP IS NOT NULL))

Example 6: Countries with more cities than the average number of cities per country in their continent
SELECT c.Name, c.Continent,
   	(SELECT COUNT(*) FROM city ci WHERE ci.CountryCode = c.Code) as CityCount
